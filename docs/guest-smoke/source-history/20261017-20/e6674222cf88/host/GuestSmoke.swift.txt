// sow: pomar-guest-SOW-01
// Isolated source preparation. No VM start, install, signer or admission API.
import Foundation
import Darwin

enum GuestSmoke {
    static let maximumFrame = 1 << 20
    static let schema = "pomar.guest-smoke/v1"
    static let checkNames: Set<String> = [
        "os", "kernel", "tools", "pid1", "settled_unit", "main_pid",
        "start_timeout", "unit_cleanup",
    ]

    enum Failure: Error, Equatable {
        case invalidBinding, invalidLimit, truncated, oversized, extraData
        case deadline, transport, invalidJSON, duplicateKey, invalidSchema
        case wrongBinding, invalidTransition, unknownCleanup
    }

    struct Binding: Equatable {
        let runID: String
        let vmID: String
        let manifestSHA256: String

        func validate() throws {
            let idCharacters = Set("abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789._-".utf8)
            for id in [runID, vmID] {
                guard (1...128).contains(id.utf8.count), id.utf8.allSatisfy({ idCharacters.contains($0) }) else {
                    throw Failure.invalidBinding
                }
            }
            guard manifestSHA256.utf8.count == 64,
                  manifestSHA256.utf8.allSatisfy({ (48...57).contains($0) || (97...102).contains($0) }) else {
                throw Failure.invalidBinding
            }
        }
    }

    struct Check: Decodable {
        let name: String
        let ok: Bool
        let detail: String
    }

    struct Report: Decodable {
        let schema: String
        let run_id: String
        let vm_id: String
        let manifest_sha256: String
        let evidence_source: String
        let outcome: String
        let checks: [Check]
    }

    // Foundation's keyed decoders alone do not reject duplicate JSON keys.
    // Scan every object first, comparing decoded keys (including escapes).
    private struct KeyScanner {
        let bytes: [UInt8]
        var index = 0

        mutating func whitespace() {
            while index < bytes.count, [9, 10, 13, 32].contains(bytes[index]) { index += 1 }
        }

        mutating func take(_ byte: UInt8) throws {
            guard index < bytes.count, bytes[index] == byte else { throw Failure.invalidJSON }
            index += 1
        }

        mutating func string() throws -> String {
            let start = index
            try take(34)
            while index < bytes.count {
                let byte = bytes[index]
                index += 1
                if byte == 34 {
                    do { return try JSONDecoder().decode(String.self, from: Data(bytes[start..<index])) }
                    catch { throw Failure.invalidJSON }
                }
                if byte == 92 {
                    guard index < bytes.count else { throw Failure.invalidJSON }
                    index += 1
                } else if byte < 32 { throw Failure.invalidJSON }
            }
            throw Failure.invalidJSON
        }

        mutating func value(depth: Int) throws {
            guard depth <= 32 else { throw Failure.invalidJSON }
            whitespace()
            guard index < bytes.count else { throw Failure.invalidJSON }
            switch bytes[index] {
            case 123:
                index += 1
                whitespace()
                var keys = Set<String>()
                if index < bytes.count, bytes[index] == 125 { index += 1; return }
                while true {
                    let key = try string()
                    guard keys.insert(key).inserted else { throw Failure.duplicateKey }
                    whitespace()
                    try take(58)
                    try value(depth: depth + 1)
                    whitespace()
                    guard index < bytes.count else { throw Failure.invalidJSON }
                    if bytes[index] == 125 { index += 1; return }
                    try take(44)
                    whitespace()
                }
            case 91:
                index += 1
                whitespace()
                if index < bytes.count, bytes[index] == 93 { index += 1; return }
                while true {
                    try value(depth: depth + 1)
                    whitespace()
                    guard index < bytes.count else { throw Failure.invalidJSON }
                    if bytes[index] == 93 { index += 1; return }
                    try take(44)
                }
            case 34:
                _ = try string()
            default:
                let start = index
                while index < bytes.count, ![9, 10, 13, 32, 44, 93, 125].contains(bytes[index]) {
                    index += 1
                }
                guard index > start else { throw Failure.invalidJSON }
            }
        }
    }

    static func decode(_ payload: Data, expected: Binding) throws -> Report {
        try expected.validate()
        guard !payload.isEmpty, payload.count <= maximumFrame else { throw Failure.oversized }
        guard String(data: payload, encoding: .utf8) != nil else { throw Failure.invalidJSON }
        var scanner = KeyScanner(bytes: Array(payload))
        try scanner.value(depth: 0)
        // No trailing whitespace or second value: the payload is exactly one object.
        guard scanner.index == payload.count else { throw Failure.extraData }
        let object: [String: Any]
        do {
            guard let parsed = try JSONSerialization.jsonObject(with: payload) as? [String: Any] else {
                throw Failure.invalidSchema
            }
            object = parsed
        } catch let failure as Failure { throw failure }
        catch { throw Failure.invalidJSON }
        let required: Set<String> = [
            "schema", "run_id", "vm_id", "manifest_sha256", "evidence_source", "outcome", "checks",
        ]
        guard Set(object.keys) == required,
              let checks = object["checks"] as? [[String: Any]], checks.count == checkNames.count,
              checks.allSatisfy({ Set($0.keys) == Set(["name", "ok", "detail"]) }) else {
            throw Failure.invalidSchema
        }
        let report: Report
        do { report = try JSONDecoder().decode(Report.self, from: payload) }
        catch { throw Failure.invalidSchema }
        guard report.schema == schema, report.evidence_source == "SELF_REPORTED",
              ["pass", "fail"].contains(report.outcome),
              Set(report.checks.map(\.name)) == checkNames,
              report.checks.allSatisfy({ !$0.detail.isEmpty && $0.detail.utf8.count <= 4096 }),
              (report.outcome == "pass") == report.checks.allSatisfy(\.ok) else {
            throw Failure.invalidSchema
        }
        guard report.run_id == expected.runID, report.vm_id == expected.vmID,
              report.manifest_sha256 == expected.manifestSHA256 else { throw Failure.wrongBinding }
        return report
    }

    // Sole reader of an already-owned connection. The caller owns descriptor
    // lifetime; no path, socket, VM, network or credential is created here.
    static func receive(fd: Int32, expected: Binding, timeout: TimeInterval = 120) throws -> Report {
        try expected.validate()
        // poll ignores negative descriptors, which otherwise disguises an
        // invalid connection as a timeout rather than a transport refusal.
        guard fd >= 0 else { throw Failure.transport }
        guard timeout.isFinite, timeout > 0, timeout <= 120 else { throw Failure.invalidLimit }
        let deadline = DispatchTime.now().uptimeNanoseconds + UInt64(timeout * 1_000_000_000)

        func readSome(_ buffer: UnsafeMutableRawBufferPointer) throws -> Int {
            while true {
                let now = DispatchTime.now().uptimeNanoseconds
                guard now < deadline else { throw Failure.deadline }
                let remaining = deadline - now
                let milliseconds = Int32(min(UInt64(Int32.max), (remaining + 999_999) / 1_000_000))
                var descriptor = pollfd(fd: fd, events: Int16(POLLIN), revents: 0)
                let ready = poll(&descriptor, 1, milliseconds)
                if ready == 0 { continue }
                if ready < 0 {
                    if errno == EINTR { continue }
                    throw Failure.transport
                }
                guard DispatchTime.now().uptimeNanoseconds < deadline else { throw Failure.deadline }
                let count = Darwin.read(fd, buffer.baseAddress, buffer.count)
                if count >= 0 { return count }
                if errno != EINTR && errno != EAGAIN && errno != EWOULDBLOCK { throw Failure.transport }
            }
        }

        func exact(_ count: Int) throws -> Data {
            var bytes = [UInt8](repeating: 0, count: count)
            var offset = 0
            while offset < count {
                let received = try bytes.withUnsafeMutableBytes { storage in
                    try readSome(UnsafeMutableRawBufferPointer(rebasing: storage[offset..<count]))
                }
                guard received > 0 else { throw Failure.truncated }
                offset += received
            }
            return Data(bytes)
        }

        let header = try exact(4)
        let count = header.reduce(UInt32(0)) { ($0 << 8) | UInt32($1) }
        // Enforce the cap before allocation or body read.
        guard count > 0, count <= UInt32(maximumFrame) else { throw Failure.oversized }
        let report = try decode(exact(Int(count)), expected: expected)
        var terminal = [UInt8](repeating: 0, count: 1)
        let extra = try terminal.withUnsafeMutableBytes { try readSome($0) }
        guard extra == 0 else { throw Failure.extraData }
        return report
    }

    // Per-run refusal model only; not a cross-process admission issuer.
    struct Lifecycle {
        enum Phase { case prepared, booting, running, stopping, quiescent, removed, quarantined }
        private(set) var phase = Phase.prepared
        let binding: Binding

        init(binding: Binding) throws {
            try binding.validate()
            self.binding = binding
        }

        mutating func beginBoot() throws {
            guard phase == .prepared else { throw Failure.invalidTransition }
            phase = .booting
        }

        mutating func booted() throws {
            guard phase == .booting else { throw Failure.invalidTransition }
            phase = .running
        }

        mutating func beginStop() throws {
            guard [.prepared, .booting, .running].contains(phase) else { throw Failure.invalidTransition }
            phase = .stopping
        }

        mutating func confirmQuiescence(vmStopped: Bool, executorReaped: Bool) throws {
            guard phase == .stopping else { throw Failure.invalidTransition }
            guard vmStopped && executorReaped else {
                phase = .quarantined
                throw Failure.unknownCleanup
            }
            phase = .quiescent
        }

        var mayRemoveRunMutableData: Bool { phase == .quiescent }

        mutating func confirmRemoval(pathsAbsent: Bool) throws {
            guard phase == .quiescent else { throw Failure.invalidTransition }
            guard pathsAbsent else {
                phase = .quarantined
                throw Failure.unknownCleanup
            }
            phase = .removed
        }
    }
}
